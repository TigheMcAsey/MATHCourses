    \section{Heegaard Diagrams of 3-Manifolds}

    \begin{example}[\(\mathbf{L(p,q) := S^3/[(w,z) \sim (\xi_p w, \xi_p^q z)]}\)]

        Consider the standard Heegaard decomposition of \(S^3\) given by 
        \begin{align}
            S^3 = \underbrace{\set{\abs{z} \geq \abs{w}}}_{h^1} \cup_{\underbrace{\abs{z} = \abs{w}}_{\mathbb{T}^2}} \underbrace{\set{\abs{w} \geq \abs{z}}}_{h^2}
        \end{align}
        Then the action of \(\mathbb{Z}/(p)\) which I will denote by \(f\) preserves the handlebodies in this decomposition alongside the torus \(\abs{z} = \abs{w}\). This gives us
        \begin{align}
            L(p,q) = h^1/f \cup_{\mathbb{T}^2/f} h^2/f \label{quot}
        \end{align}
        This is useless unless we can describe what these sets look like, notice that if we can describe \(\mathbb{T}^2/f\), then we have also described the handlebodies, since these can be expressed as \(\mathbb{T}^2/f \times I\). Computing for \(\mathbb{T}^2\):
        \begin{align}
            \mathbb{T}^2/f = \frac{\mathbb{T}^2}{(x,y) \sim \left(x + \frac{1}{p},y+\frac{q}{p}\right)} = \frac{\mathbb{R}^2/(\mathbb{Z}^2)}{(\frac{1}{p},\frac{q}{p})\mathbb{Z}} = \frac{\mathbb{R}^2}{(0,1)\mathbb{Z} \oplus (\frac{1}{p},\frac{q}{p})\mathbb{Z}} \cong \mathbb{T}^2 \label{torusbasis}
        \end{align}
        Thus equation \eqref{quot} also specifies a Heegaard Decomposition. Using the basis computed in equation \eqref{torusbasis} we take the corresponding generators for \(H_1(\partial h^j)\). Now \((0,1)\mathbb{Z}\) is the meridian for \(h^1\) and thus also \(h^1/f\), same for \((1,0)\) as the meridian of \(h^2, h^2/f\), The meridian of \(h^2\) is thus glued to \(p(1/p,q/p) - q(0,1)\). This gives the heegaard decomposition
        on \(\mathbb{T}^2\) as the curves given by \((\alpha, p\beta - q\alpha)\), where \(\alpha,\beta \in H^1(\mathbb{T}^2)\) are the homology basis of the quotient computed in equation \eqref{torusbasis}.
    \end{example}

    \begin{example}[\(\mathbb{T}^3\) with the flat metric]
        Model \(\mathbb{T}^3\) as \([0,2\pi)^3\), then consider
        \begin{align}
            f(x) = \sum_1^3 \frac12 \cos(x^j) + \frac32
        \end{align}
        I claim that this is a self-indexing morse function, checking this
        \begin{align}
            -\nabla f = \frac12 \sin(x^j)dx_j \implies (-\nabla f)^{-1}(0) = \set{(s,t,r) \mid s,t,r \in \set{0,\pi}}
        \end{align}
        And at such points we have
        \begin{align}
            \operatorname{Hess}(f) = \frac12 \operatorname{Diag}(-\cos(x^j))
        \end{align}
        This verifies \(f\) is morse, to see its self indexing note that the index is the number of occurences of \(0\) in the critical point vector.

        Now we solve for a gradient flow of \(f\), from the above computations
        \begin{align}
            \dot{x}^j = \frac12 \sin(x^j)
        \end{align}
        We can apply seperation of variables to \(y'/\sin y = \frac12\) this gives us \(\int \frac{1}{\sin(x^j)} dx^j = \frac{t}{2} + C\) integrating gives us
        \begin{align}
            \log(\tan(x^j/2)) = \frac{t}{2} + C
        \end{align}
        and thus
        \begin{align}
            x^j = 2\arctan(C e^{t/2})
        \end{align}
        Using the initial condition \(x^j(0), C = \tan(x^j(0)/2)\) so that
        \begin{align}
            x^j = 2\arctan(\tan(x^j(0)/2) e^{t/2})
        \end{align}
        The Heegaard diagram uses the asymptotic information of gradient flows, luckily this is easy to compute:
        \begin{align}
            &\lim_{t\to\infty} x^j = \begin{cases}
                \pi & x^j(0) \neq 0 \\
                0 & x^j(0) = 0
            \end{cases} &\lim_{t\to-\infty} x^j = \begin{cases}
                0 &x^j(0) \neq \pi \\
                \pi &x^j(0) = \pi
            \end{cases}
        \end{align}
        (We make sense of this by thinking of \(S^1\) as the codomain of \(\tan\) and domain of \(\arctan\))
        It follows that the stable manifold of the index one critical point \((0,\pi,\pi)\) is given by \(\set{(0,x^2,x^3) \mid x^2,x^3 \neq 0}\). The others are described symmetrically. Similarly we can consider the unstable manifold of the critical point \((\pi,0,0)\) as \(\set{(\pi,x^2,x^3) \mid x^2,x^3 \neq \pi}\), the rest of the cases are symmetric.

        Now, to get the Heegaard decomposition, we study the level set \(f^{-1}(3/2)\), the \(\alpha\) curves are given by
        \begin{align}
            &\alpha_1 = \set{(0,x^2,x^3)\mid \cos(x^2) + \cos(x^3) = -1}\\
            &\alpha_2 = \set{(x^1,0,x^3)\mid \cos(x^1) + \cos(x^3) = -1}\\
            &\alpha_3 = \set{(x^1,x^2,0)\mid \cos(x^1) + \cos(x^2) = -1}
        \end{align}
        And the \(\beta\) curves
        \begin{align}
            &\beta_1 = \set{(\pi,x^2,x^3)\mid \cos(x^2) + \cos(x^3) = 1}\\
            &\beta_2 = \set{(x^1,\pi,x^3)\mid \cos(x^1) + \cos(x^3) = 1}\\
            &\beta_3 = \set{(x^1,x^2,\pi)\mid \cos(x^1) + \cos(x^2) = 1}
        \end{align}
    \end{example}

    \section{Symmetric Products}
    \begin{theorem}[Osvath Szabo Intro to Floer 4.1]
        \(\pi_1(\operatorname{Sym}^g(\Sigma)) \cong H_1(\operatorname{Sym}^g(\Sigma)) \cong H_1(\Sigma)\)

        From Dusa McDuff's survey paper, the isomorphism is induced by inclusion \(\Sigma \hookrightarrow \Sigma \times \set{z}^{g-1}\)
    \end{theorem}

    \begin{pb}
        Compute \(\pi_2(\operatorname{Sym}^2(\Sigma_2))\)

        \begin{proof}
            As written in the notes, \(\operatorname{Sym}^2(\Sigma_2) \cong \mathbb{T}^4 \# \mathbb{CP}^2\). Then we know that \(\mathbb{CP}^2\) is given by gluing \(D^4\) to \(\mathbb{CP}^1 \cong S^2\) along \(\partial D^4 = S^3\). We can take the connect sum to be cutting out this \(D^4\) and gluing along \(S^3\). We see that if we first took \((\mathbb{T}^4,D^4)\) and \((\mathbb{CP}^4,D^4)\) as CW pairs and quotiented before gluing then we get \(\mathbb{T}^4\) with \(\mathbb{CP}^2 \cong S^2\) glued to a point, this is a homotopy equivalence by Hatcher 0.16. Thus we find that
            \begin{align}
                \operatorname{Sym}^2(\Sigma_2) \cong \mathbb{T}^4 \vee S^2
            \end{align}
            Observe that this has universal covering space \(X\) given by \(\mathbb{R}^4\) with spheres placed at each point in \(\mathbb{Z}^4\). The Hurewicz theorem tells us that \(\pi_2(X) \cong H_2(X)\), and the fibration
            \begin{align}
                \mathbb{Z}^4 \to X \to \mathbb{T}^4 \vee S^2
            \end{align}
            gives the long exact sequence of homotopy groups (Hatcher 4.2)
            \begin{equation}
                \begin{tikzcd}
                \cdots \arrow[r] &\pi_j(\mathbb{Z}^4) \arrow[r] &\pi_j(X) \arrow[r] &\pi_j(\mathbb{T}^4 \vee S^2) \arrow[r] &\pi_{j-1}(\mathbb{Z}^4) \arrow[r] &\cdots
            \end{tikzcd}
            \end{equation}
            Setting \(j\geq2\) the homotopy groups of \(\mathbb{Z}^4\) are zero so we get \(\pi_2(X) \cong \pi_2(\mathbb{T}^4 \vee S^2)\). Thus it suffices to compute \(H_2(X)\), we can see this is \(\mathbb{Z}^4\) since \(X \simeq \bigvee_{\mathbb{Z}^4}S^2\). We conclude that
            \begin{align}
                \pi_2(\operatorname{Sym}^2(\Sigma_2)) \cong \pi_2(\mathbb{T}^4 \vee S^2) \cong \bigoplus_{\mathbb{Z}^4}\mathbb{Z}
            \end{align}
        \end{proof}
    \end{pb}

    \begin{pb}
        Show that \(\mathbb{T}_\alpha\) is a totally real submanifold of \(\operatorname{Sym}^g(\Sigma)\)

        \begin{proof}
            Use abuse of notation to consider \(\mathbb{T}_\alpha \subset \Sigma^g\), then since \(\mathbb{T}_\alpha\) is a submanifold and so is the diagonal \(\Delta = (\Sigma^g)^{S_g}\) and they are disjoint, we can take a tubular neighborhood \(N\) of \(\Delta\) which is disjoint from \(\mathbb{T}_\alpha\). Moreover, by possibly shrinking \(N\) we may assume that \(S_g: N \to N\). It follows that \(S_g: \Sigma^g \setminus N \to \Sigma^g \setminus N\) acts properly discontinuously. Consider the following, the horizontal arrows are inclusions and the vertical are quotient by \(S_g\).
            \begin{equation}\begin{tikzcd}
                \mathbb{T}_\alpha \arrow[d ,"\cong"]\arrow[r,"\iota"] &\Sigma^g \setminus N \arrow[d] \arrow[r,"\iota"] &\Sigma^g \arrow[d] \\
                \mathbb{T}_\alpha \arrow[r, "\iota"] &\operatorname{Sym}^g(\Sigma) \setminus N \arrow[r, "\iota"] &\operatorname{Sym}^g(\Sigma)
            \end{tikzcd}\end{equation}
            Apply \(T_*\) to the above diagram to get
            \begin{equation}\begin{tikzcd}
                \bigoplus_j T_* \alpha_j \arrow[d, "\cong"]\arrow[r,"\iota"] &T_*(\Sigma^g \setminus N) \arrow[d] \arrow[r,"\iota"] &\bigoplus_j T_*\Sigma \arrow[d, ""] \\
                T_*\mathbb{T}_\alpha \arrow[r, "\iota"] &T_*\operatorname{Sym}^g(\Sigma) \setminus N \arrow[r, "\iota"] &T_*\operatorname{Sym}^g(\Sigma)
            \end{tikzcd}\end{equation}
            The complex structure being induced by that of \(\Sigma\) should mean that \(J(T_*\mathbb{T}_\alpha) = \bigoplus_\alpha JT\) and that \(J\) commutes with the quotient action. This reduces the question of \(\mathbb{T}_\alpha\) being totally real to showing that each of \(\alpha_j\) being totally real. This is trivial since for \((v,p) \in T_* \alpha_j\) we have \(Jv\) is linearly independent from \(v\).
        \end{proof}
    \end{pb}
